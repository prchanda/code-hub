Question: https://leetcode.com/problems/time-based-key-value-store/

Explanation: https://www.youtube.com/watch?v=fu2cD_6E8Hw

```cs
public class TimeMap
{
    private class Entry
    {
        public int Timestamp;
        public string Value;

        public Entry(int timestamp, string value)
        {
            Timestamp = timestamp;
            Value = value;
        }
    }

    private readonly Dictionary<string, List<Entry>> map;

    public TimeMap()
    {
        map = new Dictionary<string, List<Entry>>();
    }

    public void Set(string key, string value, int timestamp)
    {
        if (!map.ContainsKey(key))
        {
            map[key] = new List<Entry>();
        }

        map[key].Add(new Entry(timestamp, value));
    }

    public string Get(string key, int timestamp)
    {
        if (!map.TryGetValue(key, out List<Entry> entries))
            return "";

        int left = 0;
        int right = entries.Count - 1;

        string result = "";

        while (left <= right)
		{
		    int mid = left + (right - left) / 2;
		
		    if (entries[mid].Timestamp == timestamp)
		        return entries[mid].Value;
		
		    if (entries[mid].Timestamp < timestamp)
		    {
		        result = entries[mid].Value;
		        left = mid + 1;
		    }
		    else
		    {
		        right = mid - 1;
		    }
		}
		
		return result;
    }
}
```

|Operation|Time|Why|
|---|--:|---|
|`Set()`|**O(1)**|Timestamps are strictly increasing, so we simply append|
|`Get()`|**O(log N)**|Binary search over timestamps for that key|
|Constructor|**O(1)**|Initialize dictionary|
|Space|**O(N)**|Store all key-value-timestamp entries|
