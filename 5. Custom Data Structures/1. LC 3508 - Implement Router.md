```cs
public class Router 
{
    private readonly int memoryLimit;
    private readonly Queue<(int Source, int Destination, int Timestamp)> queue = new();
    private readonly HashSet<(int Source, int Destination, int Timestamp)> packets = new();
    private readonly Dictionary<int, List<int>> timestamps = new();
    private readonly Dictionary<int, int> startIndex = new();

    public Router(int memoryLimit) 
    {
        this.memoryLimit = memoryLimit;
    }

    public bool AddPacket(int source, int destination, int timestamp) 
    {
        var packet = (source, destination, timestamp);

        if (!packets.Add(packet))
            return false;

        if (queue.Count >= memoryLimit) 
        {
            ForwardPacket();
        }

        queue.Enqueue(packet);

        if (!timestamps.TryGetValue(destination, out var list)) 
        {
            list = new List<int>();
            timestamps[destination] = list;
            startIndex[destination] = 0;
        }

        list.Add(timestamp);
        return true;
    }

    public int[] ForwardPacket() 
    {
        if (queue.Count == 0)
            return Array.Empty<int>();

        var packet = queue.Dequeue();
        packets.Remove(packet);

        int dest = packet.Destination;
        startIndex[dest]++;

        return new int[] { packet.Source, packet.Destination, packet.Timestamp };
    }

    public int GetCount(int destination, int startTime, int endTime) 
    {
        if (!timestamps.TryGetValue(destination, out var list))
            return 0;

        int start = startIndex[destination];
        int left = LowerBound(list, start, startTime);
        int right = UpperBound(list, start, endTime);

        return right - left;
    }

    private int LowerBound(List<int> list, int start, int target) 
    {
        int left = start, right = list.Count;
        while (left < right) 
        {
            int mid = left + (right - left) / 2;
            if (list[mid] >= target)
                right = mid;
            else
                left = mid + 1;
        }
        return left;
    }

    private int UpperBound(List<int> list, int start, int target) 
    {
        int left = start, right = list.Count;
        while (left < right) 
        {
            int mid = left + (right - left) / 2;
            if (list[mid] > target)
                right = mid;
            else
                left = mid + 1;
        }
        return left;
    }
}
```

### Complexity

- **Time Complexity:**
    
    - **`AddPacket`:** $\mathcal{O}(1)$ average time, driven by hash set insertions, queue operations, and dictionary lookups.
        
    - **`ForwardPacket`:** $\mathcal{O}(1)$ time, performing standard queue dequeue and set/dictionary updates.
        
    - **`GetCount`:** $\mathcal{O}(\log k)$ time, where $k$ is the number of timestamps recorded for that specific destination, using binary search (`LowerBound`/`UpperBound`).
        
- **Space Complexity:** $\mathcal{O}(N)$ where $N$ is the maximum number of active packets stored concurrently across the queue, hash set, and destination lists.
