Question: https://leetcode.com/problems/lfu-cache/description/

```cs
public class LFUCache
{
    private class Node
    {
        public int Key;
        public int Value;
        public int Frequency;

        public Node(int key, int value)
        {
            Key = key;
            Value = value;
            Frequency = 1;
        }
    }

    private readonly int capacity;
    private int minFrequency;

    // key -> LinkedListNode
    private readonly Dictionary<int, LinkedListNode<Node>> cache;

    // frequency -> LRU list
    private readonly Dictionary<int, LinkedList<Node>> frequencyLists;

    public LFUCache(int capacity)
    {
        this.capacity = capacity;
        cache = new Dictionary<int, LinkedListNode<Node>>();
        frequencyLists = new Dictionary<int, LinkedList<Node>>();
    }

    public int Get(int key)
    {
        if (!cache.TryGetValue(key, out LinkedListNode<Node> node))
            return -1;

        IncreaseFrequency(node);

        return node.Value.Value;
    }

    public void Put(int key, int value)
    {
        // Key already exists
        if (cache.TryGetValue(key, out LinkedListNode<Node> node))
        {
            node.Value.Value = value;
            IncreaseFrequency(node);
            return;
        }

        // Cache is full → evict LFU
        if (cache.Count == capacity)
        {
            LinkedList<Node> list = frequencyLists[minFrequency];

            LinkedListNode<Node> nodeToRemove = list.Last;

            list.RemoveLast();
            cache.Remove(nodeToRemove.Value.Key);
        }

        // Add new node with frequency = 1
        Node newNode = new Node(key, value);

        if (!frequencyLists.ContainsKey(1))
            frequencyLists[1] = new LinkedList<Node>();

        LinkedListNode<Node> newListNode = frequencyLists[1].AddFirst(newNode);

        cache[key] = newListNode;

        minFrequency = 1;
    }

    private void IncreaseFrequency(LinkedListNode<Node> node)
    {
        Node current = node.Value;
        int oldFrequency = current.Frequency;

        LinkedList<Node> oldList = frequencyLists[oldFrequency];

        // Remove from old frequency's LRU list
        oldList.Remove(node);

        // If this was the only node in minFrequency,
        // minFrequency increases.
        if (oldFrequency == minFrequency && oldList.Count == 0)
        {
            minFrequency++;
        }

        // Move node to next frequency
        current.Frequency++;

        int newFrequency = current.Frequency;

        if (!frequencyLists.ContainsKey(newFrequency))
            frequencyLists[newFrequency] = new LinkedList<Node>();

        LinkedListNode<Node> newNode = frequencyLists[newFrequency].AddFirst(current);

        // Dictionary must now point to the node
        // in the new frequency list.
        cache[current.Key] = newNode;
    }
}
```


- **Time Complexity:**
    
    - `Get(key)`: $O(1)$ amortized/worst-case due to hash map lookup and constant-time doubly linked list operations.
        
    - `Put(key, value)`: $O(1)$ amortized/worst-case since insertions, hash map lookups, and evictions all take constant time.
        
- **Space Complexity:** $O(capacity)$ to store the nodes in the hash map and frequency doubly linked lists up to the specified capacity limit.
