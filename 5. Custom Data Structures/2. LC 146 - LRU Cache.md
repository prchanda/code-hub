```cs
public class LRUCache 
{
    private readonly int capacity;

    // Hash map for O(1) lookups: maps the key directly to its node reference in the linked list
    private readonly Dictionary<int, LinkedListNode<KeyValuePair<int, int>>> cache;

    // Doubly linked list maintaining usage order: head = most recently used, tail = least recently used
    private readonly LinkedList<KeyValuePair<int, int>> lruList;

    public LRUCache(int capacity) 
    {
        this.capacity = capacity;
        // Pre-allocate capacity to prevent dictionary resizing overhead
        cache = new Dictionary<int, LinkedListNode<KeyValuePair<int, int>>>(capacity);
        lruList = new LinkedList<KeyValuePair<int, int>>();
    }

    public int Get(int key) 
    {
        // If the key is not in the cache, return -1
        if (!cache.TryGetValue(key, out var node))
            return -1;

        // Move the accessed node to the front (head) of the list to mark it as most recently used
        lruList.Remove(node);
        lruList.AddFirst(node);

        return node.Value.Value;
    }

    public void Put(int key, int value) 
    {
        // If the key already exists, update its value and move it to the front
        if (cache.TryGetValue(key, out var node)) 
        {
            node.Value = new KeyValuePair<int, int>(key, value);

            lruList.Remove(node);
            lruList.AddFirst(node);
            return;
        }

        // If the cache has reached capacity, evict the least recently used item from the tail
        if (cache.Count == capacity) 
        {
            var lruNode = lruList.Last;

            cache.Remove(lruNode.Value.Key);
            lruList.RemoveLast();
        }

        // Create a new node and place it at the front (most recently used position)
        var newNode = new LinkedListNode<KeyValuePair<int, int>>(
            new KeyValuePair<int, int>(key, value)
        );

        lruList.AddFirst(newNode);
        cache[key] = newNode;
    }
}
```
