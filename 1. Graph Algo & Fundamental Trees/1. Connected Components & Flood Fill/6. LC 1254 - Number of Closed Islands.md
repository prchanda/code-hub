Question: https://leetcode.com/problems/number-of-closed-islands/

```cs
public class Solution {
    public int ClosedIsland(int[][] grid) {
        if (grid == null || grid.Length == 0 || grid[0].Length == 0)
            return 0;

        int rows = grid.Length;
        int cols = grid[0].Length;
        int count = 0;

        // 1. Flood all border-connected land (0s -> 1s) from left and right boundaries
        for (int row = 0; row < rows; row++)
        {
            DFS(grid, row, 0, rows, cols);
            DFS(grid, row, cols - 1, rows, cols);
        }

        // 2. Flood all border-connected land (0s -> 1s) from top and bottom boundaries
        for (int col = 0; col < cols; col++)
        {
            DFS(grid, 0, col, rows, cols);
            DFS(grid, rows - 1, col, rows, cols);
        }

        // 3. Count remaining enclosed islands (0s) and flood each one visited
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (grid[row][col] == 0)
                {
                    count++;   
                    DFS(grid, row, col, rows, cols);
                }
            }
        }

        return count;
    }

    private void DFS(int[][] grid, int row, int col, int rows, int cols)
    {
        // Base Condition: Out of bounds OR water/already-visited cell (1)
        if (row < 0 || row >= rows || col < 0 || col >= cols || grid[row][col] == 1)
            return;

        grid[row][col] = 1; // Sink land to water

        // Explore 4 adjacent cells
        DFS(grid, row - 1, col, rows, cols);
        DFS(grid, row + 1, col, rows, cols);
        DFS(grid, row, col - 1, rows, cols);
        DFS(grid, row, col + 1, rows, cols);
    }
}
```


|**Metric**|**Complexity**|**Explanation**|
|---|---|---|
|**Time Complexity**|$O(m \times n)$|Every cell is visited a constant number of times across the border sweeps and the main grid scan.|
|**Space Complexity**|$O(m \times n)$|Worst-case call stack depth reaches $m \times n$ (e.g., a serpentine path of land `0`s).|
