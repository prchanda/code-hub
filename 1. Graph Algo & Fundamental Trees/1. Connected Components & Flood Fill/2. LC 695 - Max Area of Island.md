Question: https://leetcode.com/problems/max-area-of-island/

**Approach 1: DFS Solution** (Recursion Stack)

```cs
public class Solution {
    public int MaxAreaOfIsland(int[][] grid) {
        // Guard clause for null or empty grid
        if (grid == null || grid.Length == 0 || grid[0].Length == 0)
            return 0;

        int rows = grid.Length;
        int cols = grid[0].Length;
        int maxArea = 0;

        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (grid[row][col] == 1)
                {
                    int area = DFS(grid, row, col, rows, cols);
                    maxArea = Math.Max(maxArea, area);
                }
            }
        }

        return maxArea;
    }

    private int DFS(int[][] grid, int row, int col, int rows, int cols)
    {
        // Base condition: bounds check or water (0) / visited cell
        if (row < 0 || row >= rows || col < 0 || col >= cols || grid[row][col] == 0)
            return 0;

        // Mark cell as visited by sinking land (1 -> 0)
        grid[row][col] = 0;

        // Sum current cell (1) + all 4 adjacent cells
        return 1 + DFS(grid, row - 1, col, rows, cols)
                 + DFS(grid, row + 1, col, rows, cols)
                 + DFS(grid, row, col - 1, rows, cols)
                 + DFS(grid, row, col + 1, rows, cols);
    }
}
```


## Complexity Analysis

| **Metric**           | **Complexity**  | **Details**                                                                                                                |
| -------------------- | --------------- | -------------------------------------------------------------------------------------------------------------------------- |
| **Time Complexity**  | $O(m \times n)$ | Each cell is visited at most a constant number of times (once in the outer loops, plus up to 4 recursive neighbor checks). |
| **Space Complexity** | $O(m \times n)$ | In the worst case (a grid filled entirely with `1`s), the call stack depth will reach $m \times n$.                        |
