Question: https://leetcode.com/problems/surrounded-regions/description/

```cs
public class Solution {
    public void Solve(char[][] board) {
        if (board == null || board.Length == 0 || board[0].Length == 0)
            return;

        int rows = board.Length;
        int cols = board[0].Length;

        // 1. Traverse left and right borders
        for (int row = 0; row < rows; row++)
        {
            DFS(board, row, 0, rows, cols);
            DFS(board, row, cols - 1, rows, cols);
        }

        // 2. Traverse top and bottom borders
        for (int col = 0; col < cols; col++)
        {
            DFS(board, 0, col, rows, cols);
            DFS(board, rows - 1, col, rows, cols);
        }

        // 3. Single-pass cleanup: Flip 'O' -> 'X' and 'Z' -> 'O'
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (board[row][col] == 'O')
                {
                    board[row][col] = 'X';
                }
                else if (board[row][col] == 'Z')
                {
                    board[row][col] = 'O';
                }
            }
        }
    }

    private void DFS(char[][] board, int row, int col, int rows, int cols)
    {
        if (row < 0 || row >= rows || col < 0 || col >= cols || board[row][col] != 'O')
            return;

        board[row][col] = 'Z'; // Mark as safe

        // Explore 4 adjacent cells
        DFS(board, row - 1, col, rows, cols);
        DFS(board, row + 1, col, rows, cols);
        DFS(board, row, col - 1, rows, cols);
        DFS(board, row, col + 1, rows, cols);
    }
}
```




## Complexity Analysis

|**Metric**|**Complexity**|**Details**|
|---|---|---|
|**Time Complexity**|$O(m \times n)$|Every cell is visited a constant number of times across the border scans and matrix sweeps.|
|**Space Complexity**|$O(m \times n)$|In the worst-case (e.g., all `'O'`s), the recursive call stack depth reaches $m \times n$.|
