Question: https://leetcode.com/problems/count-sub-islands/

```cs
public class Solution 
{
    public int CountSubIslands(int[][] grid1, int[][] grid2) 
    {
        int rows = grid1.Length;
        int cols = grid1[0].Length;
        int subIslands = 0;

        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (grid2[row][col] == 1)
                {
                    subIslands += Dfs(grid1, grid2, row, col) ? 1 : 0;
                }
            }
        }

        return subIslands;
    }

    private bool Dfs(int[][] grid1, int[][] grid2, int row, int col)
    {
        int rows = grid1.Length;
        int cols = grid1[0].Length;

        // Base case: Out of bounds or water cell in grid2
        /*
	        grid2:
			1 1
			1 1
			
			When it tries: DFS(-1, 0), that's outside the grid.
			There is no violation here. The fact that we reached the boundary doesn't mean the island is invalid.
		*/ 
        if (row < 0 || row >= rows || col < 0 || col >= cols || grid2[row][col] != 1)
        {
            return true;
        }

        // Check if current cell in grid1 is land
        bool isSubIsland = grid1[row][col] == 1;

        // Mark cell as visited (sink the island in grid2)
        grid2[row][col] = 0;

        // Explore all 4 directions using non-short-circuiting logical AND (&)
        // to ensure all connected land cells in grid2 are visited and sunk.
        isSubIsland &= Dfs(grid1, grid2, row - 1, col)
                     & Dfs(grid1, grid2, row + 1, col)
                     & Dfs(grid1, grid2, row, col - 1)
                     & Dfs(grid1, grid2, row, col + 1);

        return isSubIsland;
    }
}
```


|**Metric**|**Complexity**|**Details**|
|---|---|---|
|**Time Complexity**|$O(m \times n)$|**Phase 1 (DFS):** Every cell is visited at most once.<br><br>  <br><br>**Phase 2 (Check `0`s):** Every `0` checks 4 constant neighbors with $O(1)$ HashSet lookups.|
|**Space Complexity**|$O(m \times n)$|Recursion stack depth + `islandArea` dictionary storage for up to $m \times n$ cells.|
