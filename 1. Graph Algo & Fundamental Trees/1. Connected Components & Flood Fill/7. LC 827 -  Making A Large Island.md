Question: https://leetcode.com/problems/making-a-large-island/

```cs
public class Solution {
    public int LargestIsland(int[][] grid) {
        if (grid == null || grid.Length == 0 || grid[0].Length == 0)
            return 0;

        int rows = grid.Length;
        int cols = grid[0].Length;

        Dictionary<int, int> islandArea = new Dictionary<int, int>();
        int islandId = 2; // IDs start at 2 (0 = water, 1 = unvisited land)
        int maxArea = 0;
        bool hasZero = false;

        // 1. Label each connected island with a unique ID and record its area 
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (grid[row][col] == 1)
                {
                    islandArea[islandId] = 0;
                    DFS(grid, row, col, rows, cols, islandId, islandArea);
                    maxArea = Math.Max(maxArea, islandArea[islandId]);
                    islandId++;
                }
            }
        }

        int[][] directions = new int[][] {
            new int[] { -1, 0 },
            new int[] { 1, 0 },
            new int[] { 0, -1 },
            new int[] { 0, 1 }
        };

        // 2. Evaluate every 0 to see the max combined area if flipped to 1
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (grid[row][col] == 0)
                {
                    hasZero = true;
                    int possibleArea = 1; // 1 for flipping this 0 to 1
                    HashSet<int> neighbouringIslands = new HashSet<int>();

                    foreach (var dir in directions)
                    {
                        int nr = row + dir[0];
                        int nc = col + dir[1];

                        if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] > 1)
                        {
                            neighbouringIslands.Add(grid[nr][nc]);
                        }
                    }

                    foreach (int id in neighbouringIslands)
                    {
                        possibleArea += islandArea[id];
                    }

                    maxArea = Math.Max(maxArea, possibleArea);
                }
            }
        }

        // If no '0' exists, the grid is already entirely land
        return hasZero ? maxArea : rows * cols;
    }

    private void DFS(int[][] grid, int row, int col, int rows, int cols, int islandId, Dictionary<int, int> islandArea)
    {
        if (row < 0 || row >= rows || col < 0 || col >= cols || grid[row][col] != 1)
            return;

        grid[row][col] = islandId; // Assign island ID
        islandArea[islandId]++;     // Increment area counter

        // Explore 4 directions
        DFS(grid, row - 1, col, rows, cols, islandId, islandArea);
        DFS(grid, row + 1, col, rows, cols, islandId, islandArea);
        DFS(grid, row, col - 1, rows, cols, islandId, islandArea);
        DFS(grid, row, col + 1, rows, cols, islandId, islandArea);
    }
}
```



|**Metric**|**Complexity**|**Details**|
|---|---|---|
|**Time Complexity**|$O(m \times n)$|**Phase 1 (DFS):** Every cell is visited at most once.<br><br>  <br><br>**Phase 2 (Check `0`s):** Every `0` checks 4 constant neighbors with $O(1)$ HashSet lookups.|
|**Space Complexity**|$O(m \times n)$|Recursion stack depth + `islandArea` dictionary storage for up to $m \times n$ cells.|
