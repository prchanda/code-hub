Question: https://leetcode.com/problems/number-of-enclaves/

```cs
public class Solution {
    public int NumEnclaves(int[][] grid) {
        // Guard clause for null or empty grid
        if (grid == null || grid.Length == 0 || grid[0].Length == 0)
            return 0;

        int rows = grid.Length;
        int cols = grid[0].Length;
        int count = 0;

        // 1. Traverse left and right borders to mark reachable land as -1
        for (int row = 0; row < rows; row++)
        {
            DFS(grid, row, 0, rows, cols);
            DFS(grid, row, cols - 1, rows, cols);
        }

        // 2. Traverse top and bottom borders to mark reachable land as -1
        for (int col = 0; col < cols; col++)
        {
            DFS(grid, 0, col, rows, cols);
            DFS(grid, rows - 1, col, rows, cols);
        }

        // 3. Count all remaining trapped land cells (1s)
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (grid[row][col] == 1)
                {
                    count++;
                }
            }
        }

        return count;
    }

    private void DFS(int[][] grid, int row, int col, int rows, int cols)
    {
        // Base Condition: Out of bounds OR cell is not land (0 or already -1)
        if (row < 0 || row >= rows || col < 0 || col >= cols || grid[row][col] != 1)
            return;

        grid[row][col] = -1; // Mark as visited border-connected land

        // Explore 4-directional neighbors
        DFS(grid, row - 1, col, rows, cols);
        DFS(grid, row + 1, col, rows, cols);
        DFS(grid, row, col - 1, rows, cols);
        DFS(grid, row, col + 1, rows, cols);
    }
}
```

## Complexity Analysis

| **Metric**           | **Complexity**  | **Explanation**                                                                             |
| -------------------- | --------------- | ------------------------------------------------------------------------------------------- |
| **Time Complexity**  | $O(m \times n)$ | Every cell is evaluated a constant number of times (border checks + DFS + final grid scan). |
| **Space Complexity** | $O(m \times n)$ | In the worst case (e.g., all `1`s), the recursion stack reaches $m \times n$ depth.         |
