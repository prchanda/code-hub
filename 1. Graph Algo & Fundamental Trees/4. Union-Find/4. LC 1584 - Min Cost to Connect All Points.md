Question: https://leetcode.com/problems/min-cost-to-connect-all-points/description/

```cs
public class Solution
{
    public int MinCostConnectPoints(int[][] points)
    {
        List<List<int>> edges = new List<List<int>>();
        int[] parent = new int[points.Length];
        int[] rank = new int[points.Length];

        for (int index = 0; index < points.Length; index++)
            parent[index] = index;

        for (int index = 0; index < points.Length; index++)
        {
            for (int otherIndex = index + 1; otherIndex < points.Length; otherIndex++)
            {
                int distance = Math.Abs(points[index][0] - points[otherIndex][0]) + Math.Abs(points[index][1] - points[otherIndex][1]);
                edges.Add(new List<int> { index, otherIndex, distance });
            }
        }

        edges.Sort((a, b) => a[2].CompareTo(b[2]));
        int minimumWeight = 0;
        int edgesUsed = 0;

        foreach (var edge in edges)
        {
            int from = edge[0];
            int to = edge[1];
            int weight = edge[2];

            if (Union(from, to, parent, rank))
            {
                minimumWeight += weight;
                edgesUsed++;
            }

            if (edgesUsed == points.Length - 1)
                break;
        }

        return minimumWeight;
    }

    private bool Union(int nodeA, int nodeB, int[] parent, int[] rank)
    {
        int rootA = Find(nodeA, parent);
        int rootB = Find(nodeB, parent);

        if (rootA == rootB)
            return false;

        if (rank[rootA] < rank[rootB])
        {
            parent[rootA] = rootB;
        }
        else if (rank[rootB] < rank[rootA])
        {
            parent[rootB] = rootA;
        }
        else
        {
            parent[rootB] = rootA;
            rank[rootA]++;
        }

        return true;
    }

    private int Find(int node, int[] parent)
    {
        if (parent[node] != node)
            parent[node] = Find(parent[node], parent);

        return parent[node];
    }
}
```


### **Complexity Analysis**

- **Time Complexity:** $O(V^2 \log V)$ where $V$ is the number of points (vertices).
    
    - **Edge Generation:** Generating all pairs of points creates $\frac{V(V-1)}{2}$ edges, which takes $O(V^2)$ time.
        
    - **Sorting:** Sorting the list of edges takes $O(E \log E)$ time. Since $E = O(V^2)$, sorting takes $O(V^2 \log(V^2)) = O(V^2 \log V)$.
        
    - **Union-Find:** Processing the edges with path compression and union by rank takes nearly constant time per edge, $O(E \alpha(V))$ where $\alpha$ is the Inverse Ackermann function.
        
- **Space Complexity:** $O(V^2)$ to store all the generated edges in the list, along with $O(V)$ auxiliary space for the `parent`, `rank`, and Union-Find recursion stack.
