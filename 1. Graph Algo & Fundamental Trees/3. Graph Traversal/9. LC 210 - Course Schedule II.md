Question : https://leetcode.com/problems/course-schedule-ii/

```cs

public class Solution 
{
    private const int Unvisited = 0;
    private const int Visiting = 1;
    private const int Visited = 2;

    public int[] FindOrder(int numCourses, int[][] prerequisites) 
    {
        List<int>[] graph = new List<int>[numCourses];

        for (int i = 0; i < numCourses; i++)
        {
            graph[i] = new List<int>();
        }

        // Build directed graph: prerequisite -> course
        foreach (var edge in prerequisites)
        {
            int course = edge[0];
            int pre = edge[1];

            graph[pre].Add(course);
        }

        int[] states = new int[numCourses];
        List<int> result = new List<int>();

        // Check for cycles and perform topological sort from each course
        for (int i = 0; i < numCourses; i++)
        {
            if (HasCycle(graph, i, states, result))
            {
                return Array.Empty<int>();
            }
        }

        // Reverse post-order result to get correct prerequisite ordering
        result.Reverse();
        return result.ToArray();
    }

    private bool HasCycle(List<int>[] graph, int node, int[] states, List<int> result)
    {
        // Cycle detected: node is in current traversal path
        if (states[node] == Visiting)
            return true;

        // Fully processed node
        if (states[node] == Visited)
            return false;
	
        states[node] = Visiting;

        foreach (var neighbor in graph[node])
        {
            if (HasCycle(graph, neighbor, states, result))
                return true;
        }

        states[node] = Visited;
        result.Add(node); // Post-order collection
        return false;
    }
}

```


| **Metric**           | **Complexity** | **Details**                                                                                                                                                                                                                                                                                                                                                                                                                                                 |
| -------------------- | -------------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Time Complexity**  | $O(V + E)$     | **Graph Construction:** Scans $V$ courses and $E$ prerequisite pairs to construct the directed adjacency list in $O(V + E)$ time.<br><br>**DFS Traversal:** Each node transitions through state $0 \to 1 \to 2$ exactly once ($O(V)$). Every directed edge is explored at most once ($O(E)$).<br><br>**Result Reversal:** Reversing the $V$-element result list and converting to an array takes $O(V)$ time.<br><br>Overall time complexity is $O(V + E)$. |
| **Space Complexity** | $O(V + E)$     | **Adjacency List:** Stores $V$ node entries and $E$ total directed edges ($O(V + E)$ space).<br><br>**States & Result Storage:** The `states` array takes $O(V)$ space and the `result` list stores $V$ elements ($O(V)$ space).  <br><br>**Recursion Stack:** In the worst-case (a linear dependency chain), the DFS recursion call stack reaches a depth of $O(V)$.<br><br>Therefore, total auxiliary space is $O(V + E)$.                                |
