public class Solution
{
    public bool CanVisitAllRooms(IList<IList<int>> rooms)
    {
        int numberOfRooms = rooms.Count;
        bool[] visited = new bool[numberOfRooms];

        DFS(rooms, 0, visited);

        for (int index = 0; index < numberOfRooms; index++)
        {
            if (!visited[index])
                return false;
        }

        return true;
    }

    private void DFS(IList<IList<int>> graph, int room, bool[] visited)
    {
        visited[room] = true;
        var roomKeys = graph[room];

        foreach (var roomKey in roomKeys)
        {
            if (!visited[roomKey])
                DFS(graph, roomKey, visited);
        }
    }
}
