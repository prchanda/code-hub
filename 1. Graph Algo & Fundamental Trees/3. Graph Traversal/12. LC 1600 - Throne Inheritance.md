Question - https://leetcode.com/problems/throne-inheritance/description/

```cs
public class Person
{
    public string Name;
    public bool IsAlive;
    public List<Person> Children;

    public Person(string name)
    {
        Name = name;
        IsAlive = true;
        Children = new List<Person>();
    }
}

public class ThroneInheritance
{
    private readonly Person root;
    // Name → Person object, without the dictionary, we'd have to traverse the entire tree looking for the person. 
    private readonly Dictionary<string, Person> people;

    public ThroneInheritance(string kingName)
    {
        root = new Person(kingName);
        people = new Dictionary<string, Person>();

        people[kingName] = root;
    }

    public void Birth(string parentName, string childName)
    {
        Person parent = people[parentName];

        Person child = new Person(childName);

        parent.Children.Add(child);
        people[childName] = child;
    }

    public void Death(string name)
    {
        people[name].IsAlive = false;
    }

    public IList<string> GetInheritanceOrder()
    {
        IList<string> result = new List<string>();

        DFS(root, result);

        return result;
    }

    private void DFS(Person person, IList<string> result)
    {
        if (person.IsAlive)
        {
            result.Add(person.Name);
        }

        foreach (Person child in person.Children)
        {
            DFS(child, result);
        }
    }
}
```
