You have a graph of `n` nodes labeled from `0` to `n - 1`. You are given an integer n and a list of `edges` where `edges[i] = [ai, bi]` indicates that there is an undirected edge between nodes `ai` and `bi` in the graph.

Return `true` _if the edges of the given graph make up a valid tree, and_ `false` _otherwise_.

```cs
public class Solution
{
    public bool ValidTree(int n, int[][] edges)
    {
        // A tree with n nodes must have exactly n - 1 edges.
        if (edges.Length != n - 1)
            return false;

        // Build adjacency list
        List<int>[] graph = new List<int>[n];

        for (int i = 0; i < n; i++)
        {
            graph[i] = new List<int>();
        }

        foreach (int[] edge in edges)
        {
            int a = edge[0];
            int b = edge[1];

            graph[a].Add(b);
            graph[b].Add(a);
        }

        bool[] visited = new bool[n];

        // Start DFS from node 0.
        if (HasCycle(graph, 0, -1, visited))
            return false;

        // Tree must contain every node.
        for (int i = 0; i < n; i++)
        {
            if (!visited[i])
                return false;
        }

        return true;
    }

    private bool HasCycle(
        List<int>[] graph,
        int node,
        int parent,
        bool[] visited)
    {
        visited[node] = true;

        foreach (int neighbor in graph[node])
        {
            if (!visited[neighbor])
            {
                if (HasCycle(graph, neighbor, node, visited))
                    return true;
            }
            else if (neighbor != parent)
            {
                // Already visited, and it's not our parent.
                // Therefore, we found a cycle.
                return true;
            }
        }

        return false;
    }
}
```


| **Metric**           | **Complexity** | **Details**                                                                                                                                                                                                                                                                        |
| -------------------- | -------------- | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Time Complexity**  | $O(V + E)$     | **Edge Count Check:** $O(1)$ if you check `if (edges.Length != n - 1) return false;` first.<br><br>**Graph Building:** $O(V + E)$ to construct the adjacency list.  <br><br>**Traversal:** Visits each vertex ($O(V)$) and explores each edge ($O(E)$) once to check connectivity. |
| **Space Complexity** | $O(V + E)$     | **Adjacency List:** Takes $O(V + E)$ space to store the graph structure.  <br><br>**Visited Set & Stack/Queue:** Uses $O(V)$ space for the `visited` array and recursion call stack or BFS queue.                                                                                  |
