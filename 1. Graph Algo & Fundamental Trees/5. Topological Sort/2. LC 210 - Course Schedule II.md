Question : https://leetcode.com/problems/course-schedule-ii/

### Kahn's Algorithm

```cs
public int[] FindOrder(int numCourses, int[][] prerequisites) {
	List<int>[] graph = new List<int>[numCourses];
	int processedCourses = 0;
	Queue<int> queue = new Queue<int>();
	
	for(int index=0; index<numCourses; index++)
	{
		graph[index] = new List<int>();
	}
	
	int[] indegrees = new int[numCourses];
	foreach(int[] prerequisite in prerequisites)
	{
		int course = prerequisite[0];
		int coursePrerequisite = prerequisite[1];
		graph[coursePrerequisite].Add(course);
		indegrees[course]++;
	}
	
	for(int index=0; index<numCourses; index++)
	{
		if(indegrees[index] == 0)
			queue.Enqueue(index);
	}
	
	List<int> result = new List<int>();
	
	while(queue.Count>0)
	{
		int course = queue.Dequeue();
		result.Add(course);
		foreach(int neighbour in graph[course])
		{
			indegrees[neighbour]--;
			if(indegrees[neighbour] == 0)
				queue.Enqueue(neighbour);
		}
	}
	
	if(result.Count != numCourses)
		return Array.Empty<int>();
	
	return result.ToArray();
}
```

## Complexity Summary

Let $V = \text{numCourses}$ and $E = \text{prerequisites.Length}$.

- **Time Complexity:** $O(V + E)$
    
    - Initializing the adjacency list takes $O(V + E)$ time.        
    - Kahn's Algorithm visits each vertex (course) and edge (prerequisite) at most once.
        
- **Space Complexity:** $O(V + E)$
    
    - The graph adjacency list stores $V$ lists and $E$ total edges.        
    - The `indegree` array and queue take $O(V)$ space.
