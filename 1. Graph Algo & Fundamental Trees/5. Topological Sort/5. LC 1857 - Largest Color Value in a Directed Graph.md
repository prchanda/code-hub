public class Solution 
{
    public int LargestPathValue(string colors, int[][] edges) 
    {
        int n = colors.Length;        
        List<int>[] graph = new List<int>[n];
        int[] indegrees = new int[n];
        
        // dp[node, color] tracks the maximum frequency of each color (0-25) ending at a given node
        int[,] dp = new int[n, 26];

        // Initialize the adjacency list for each node in the graph
        for (int index = 0; index < n; index++)
        {
            graph[index] = new List<int>();
        }

        // Build the directed graph and compute the in-degrees for every node
        foreach (var edge in edges)
        {
            int from = edge[0];
            int to = edge[1];
            graph[from].Add(to);
            indegrees[to]++;
        }

        // Initialize the queue for Kahn's topological sort with all source nodes (in-degree of 0)
        Queue<int> queue = new Queue<int>();
        for (int index = 0; index < n; index++)
        {
            if (indegrees[index] == 0)
            {
                int color = colors[index] - 'a';
                queue.Enqueue(index);
                dp[index, color] = 1; // Base case: starting node has a frequency of 1 for its own color
            }
        }

        int answer = 1;
        int processedNodes = 0;

        // Process nodes level-by-level in topological order
        while (queue.Count > 0)
        {
            int currentNode = queue.Dequeue();
            int currentNodeColor = colors[currentNode] - 'a';
            
            var neighbours = graph[currentNode];
            processedNodes++;
            
            // Propagate color counts from the current node to all of its neighbors
            foreach (var neighbour in neighbours)
            {
                int neighbourColor = colors[neighbour] - 'a';
                
                for (int color = 0; color < 26; color++)
                {
                    int prevMax = dp[currentNode, color];
                    
                    // If the neighbor's color matches the current color index, increment its frequency count
                    if (neighbourColor == color)
                    {
                        prevMax++;
                    }
                    
                    // Update the DP table and track the overall maximum color path value
                    dp[neighbour, color] = Math.Max(dp[neighbour, color], prevMax);
                    answer = Math.Max(answer, dp[neighbour, color]);
                }
                
                // Decrement the neighbor's in-degree as part of topological sorting
                indegrees[neighbour]--;
                
                // If all incoming dependencies are met, push the neighbor into the queue
                if (indegrees[neighbour] == 0)
                {
                    queue.Enqueue(neighbour);
                }
            }            
        }

        // If we didn't process all nodes, a cycle exists (return -1); otherwise, return the max path value
        return processedNodes != n ? -1 : answer;
    }
}
