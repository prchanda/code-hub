Question: https://leetcode.com/problems/course-schedule/

### Kahn's Algorithm

```cs
public class Solution
{
    public bool CanFinish(int numCourses, int[][] prerequisites)
    {
        // 1. Build adjacency list and indegree array
        List<int>[] graph = new List<int>[numCourses];
        for (int i = 0; i < numCourses; i++)
        {
            graph[i] = new List<int>();
        }

        int[] indegree = new int[numCourses];
        // Computer indegrees of each node
        foreach (int[] prerequisite in prerequisites)
        {
            int course = prerequisite[0];
            int prerequisiteCourse = prerequisite[1];

            graph[prerequisiteCourse].Add(course);
            indegree[course]++;
        }

        // 2. Enqueue all courses with zero prerequisites (indegree == 0)
        // Because those nodes dont have any dependencies, hence can be processed immediately
        Queue<int> queue = new Queue<int>();
        for (int course = 0; course < numCourses; course++)
        {
            if (indegree[course] == 0)
            {
                queue.Enqueue(course);
            }
        }

        int processedCourses = 0;

        // 3. Process courses via BFS (Kahn's Algorithm)
        // Pop the node and reduce the indegrees of the neighbour because we are processing the current node
        while (queue.Count > 0)
        {
            int course = queue.Dequeue();
            processedCourses++;

            foreach (int nextCourse in graph[course])
            {
                indegree[nextCourse]--;
                if (indegree[nextCourse] == 0)
                {
                    queue.Enqueue(nextCourse);
                }
            }
        }

        // 4. If all courses were processed, no directed cycles exist
        // If a connected graph contains cycles, then none of the nodes will have 0 indegrees
        return processedCourses == numCourses;
    }
}

```

## Complexity Summary

Let $V = \text{numCourses}$ and $E = \text{prerequisites.Length}$.

- **Time Complexity:** $O(V + E)$
    
    - Initializing the adjacency list takes $O(V + E)$ time.        
    - Kahn's Algorithm visits each vertex (course) and edge (prerequisite) at most once.
        
- **Space Complexity:** $O(V + E)$
    
    - The graph adjacency list stores $V$ lists and $E$ total edges.        
    - The `indegree` array and queue take $O(V)$ space.
