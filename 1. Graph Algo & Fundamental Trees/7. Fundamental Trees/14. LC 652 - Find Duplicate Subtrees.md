Question: https://leetcode.com/problems/find-duplicate-subtrees/description/

Give every unique subtree a unique integer ID. Then identical subtrees get the same ID.

```cs
public class Solution
{
    public IList<TreeNode> FindDuplicateSubtrees(TreeNode root)
    {
        var lookup = new Dictionary<(int val, int leftId, int rightId), int>();
        var count = new Dictionary<int, int>();

        IList<TreeNode> result = new List<TreeNode>();
        int nextId = 1;
        DFS(root, lookup, count, result, ref nextId);
        return result;
    }

    private int DFS(
        TreeNode root,
        Dictionary<(int val, int leftId, int rightId), int> lookup,
        Dictionary<int, int> count,
        IList<TreeNode> result,
        ref int nextId)
    {
        if (root == null)
            return 0;

        int leftId = DFS(root.left, lookup, count, result, ref nextId);

        int rightId = DFS(root.right, lookup, count, result, ref nextId);

        var key = (root.val, leftId, rightId);

        if (!lookup.TryGetValue(key, out int id))
        {
            id = nextId++;
            lookup[key] = id;
        }

        int occurrences = count.GetValueOrDefault(id) + 1;
        count[id] = occurrences;

        if (occurrences == 2)
            result.Add(root);

        return id;
    }
}
```
