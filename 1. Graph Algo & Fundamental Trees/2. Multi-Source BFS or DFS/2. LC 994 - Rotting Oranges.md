
Question: https://leetcode.com/problems/rotting-oranges/

```cs
public class Solution
{
    private static readonly (int dr, int dc)[] Directions =
    {
        (-1, 0), (1, 0), (0, -1), (0, 1)
    };

    public int OrangesRotting(int[][] grid)
    {
        int rows = grid.Length;
        int cols = grid[0].Length;

        Queue<(int row, int col)> queue = new();
        int fresh = 0;
        int minutes = 0;

        // Step 1: Find all initial rotten oranges and count fresh ones
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (grid[row][col] == 2)
                    queue.Enqueue((row, col));
                else if (grid[row][col] == 1)
                    fresh++;
            }
        }

        // Edge Case: No fresh oranges to rot
        if (fresh == 0)
            return 0;

        // Step 2: Multi-Source BFS
        while (queue.Count > 0 && fresh > 0)
        {
            int size = queue.Count;

            for (int i = 0; i < size; i++)
            {
                var (row, col) = queue.Dequeue();

                foreach (var (dr, dc) in Directions)
                {
                    int newRow = row + dr;
                    int newCol = col + dc;

                    // Skip out of bounds or non-fresh cells
                    if (newRow < 0 || newRow >= rows ||
                        newCol < 0 || newCol >= cols ||
                        grid[newRow][newCol] != 1)
                    {
                        continue;
                    }

                    // Rot the fresh orange and queue it
                    grid[newRow][newCol] = 2;
                    fresh--;

                    queue.Enqueue((newRow, newCol));
                }
            }

            minutes++;
        }

        return fresh == 0 ? minutes : -1;
    }
}
```




| **Metric**           | **Complexity**  | **Details**                                                                                                                                                                                                                                      |
| -------------------- | --------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| **Time Complexity**  | $O(m \times n)$ | **Phase 1 (Grid Scan):** Scans all $m \times n$ cells once to initialize the queue and count fresh oranges.<br><br>**Phase 2 (Multi-Source BFS):** Every cell is enqueued and dequeued at most once, performing $O(1)$ constant neighbor checks. |
| **Space Complexity** | $O(m \times n)$ | Queue storage for up to $m \times n$ cells in the worst-case scenario (e.g., all oranges start as rotten).                                                                                                                                       |
