
Question: https://leetcode.com/problems/pacific-atlantic-water-flow/description/

```cs
public class Solution
{
    // Define directions once as a static tuple array to eliminate heap allocations in recursion
    private static readonly (int dr, int dc)[] Directions = 
    {
        (-1, 0), (1, 0), (0, -1), (0, 1)
    };

    public IList<IList<int>> PacificAtlantic(int[][] heights)
    {
        int rows = heights.Length;
        int cols = heights[0].Length;

        bool[,] pacific = new bool[rows, cols];
        bool[,] atlantic = new bool[rows, cols];

        // Pacific borders: Top row & Left column
        for (int row = 0; row < rows; row++) 
	        DFS(heights, row, 0, pacific);
        for (int col = 0; col < cols; col++) 
	        DFS(heights, 0, col, pacific);

        // Atlantic borders: Bottom row & Right column
        for (int row = 0; row < rows; row++) 
	        DFS(heights, row, cols - 1, atlantic);
        for (int col = 0; col < cols; col++) 
	        DFS(heights, rows - 1, col, atlantic);

        IList<IList<int>> result = new List<IList<int>>();

        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                if (pacific[row, col] && atlantic[row, col])
                {
                    result.Add(new int[] { row, col });
                }
            }
        }

        return result;
    }

    private void DFS(int[][] heights, int row, int col, bool[,] visited)
    {
        // Guard clause: avoid re-processing if already visited
        if (visited[row, col]) 
	        return;

        visited[row, col] = true;

        int rows = heights.Length;
        int cols = heights[0].Length;

        foreach (var (dr, dc) in Directions)
        {
            int newRow = row + dr;
            int newCol = col + dc;

            // Bounds check
            if (newRow < 0 || newRow >= rows || newCol < 0 || newCol >= cols)
                continue;

            // Reverse traversal rule: next cell height must be >= current cell height
            if (heights[newRow][newCol] < heights[row][col])
                continue;

            DFS(heights, newRow, newCol, visited);
        }
    }
}
```




| **Metric**           | **Complexity**  | **Details**                                                                                                                                                                                                               |
| -------------------- | --------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Time Complexity**  | $O(m \times n)$ | **Phase 1 (DFS):** Traversing from Pacific and Atlantic borders visits each cell at most once per ocean.<br><br>**Phase 2 (Result Collection):** Iterates over all $m \times n$ cells to find overlap with $O(1)$ lookup. |
| **Space Complexity** | $O(m \times n)$ | Recursion stack depth + `pacific` / `atlantic` boolean arrays for up to $m \times n$ cells.                                                                                                                               |
