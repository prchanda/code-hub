```cs
public class Solution 
{
    public IList<IList<string>> SolveNQueens(int n) 
    {
        HashSet<int> columns = new HashSet<int>();
        HashSet<int> positiveDiagonals = new HashSet<int>();
        HashSet<int> negativeDiagonals = new HashSet<int>();
        IList<IList<string>> result = new List<IList<string>>();

        char[][] board = new char[n][];
        for (int row = 0; row < n; row++)
        {
            board[row] = new char[n];
            for (int col = 0; col < n; col++)
            {
                board[row][col] = '.';
            }
        }

        DFS(0, board, n, columns, positiveDiagonals, negativeDiagonals, result);

        return result;
    }

    private void DFS(
        int row, 
        char[][] board, 
        int n, 
        HashSet<int> columns, 
        HashSet<int> positiveDiagonals, 
        HashSet<int> negativeDiagonals,
        IList<IList<string>> result)
    {
        if (row == n)
        {
            List<string> solution = new List<string>();
            for (int r = 0; r < n; r++)
            {
                solution.Add(new string(board[r]));
            }

            result.Add(solution);
            return;
        }

        for (int col = 0; col < n; col++)
        {
            if (columns.Contains(col) || 
                positiveDiagonals.Contains(row + col) || 
                negativeDiagonals.Contains(row - col))
            {
                continue;
            }

            board[row][col] = 'Q';
            columns.Add(col);
            positiveDiagonals.Add(row + col);
            negativeDiagonals.Add(row - col);

            DFS(row + 1, board, n, columns, positiveDiagonals, negativeDiagonals, result);

            board[row][col] = '.';
            columns.Remove(col);
            positiveDiagonals.Remove(row + col);
            negativeDiagonals.Remove(row - col);
        }        
    }
}
```


**Time Complexity**

- **$O(N!)$** upper bound (or **$O(N! \cdot N^2)$** accounting for building solution strings) where $N$ is the board dimension.
    
    - In row $0$ there are $N$ choices, row $1$ has at most $N-2$ choices, row $2$ at most $N-4$ choices, leading to an upper bound governed by $O(N!)$.
        
    - When a valid placement is found, creating $N$ strings of length $N$ takes $O(N^2)$ work.
        

**Space Complexity**

- **$O(N^2)$** auxiliary space (or **$O(S \cdot N^2)$** total space including output, where $S$ is the number of valid solutions).
    
    - **$O(N^2)$** space for the 2D `board` array.
        
    - **$O(N)$** space for the `columns`, `positiveDiagonals`, and `negativeDiagonals` hash sets, plus the $O(N)$ depth of the recursion call stack.
