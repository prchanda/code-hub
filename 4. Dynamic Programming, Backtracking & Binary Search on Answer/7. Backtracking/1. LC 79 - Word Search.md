Question: https://leetcode.com/problems/word-search/description/

Explanation: https://www.youtube.com/watch?v=pfiQ_PS1g8E

```cs
public class Solution 
{
    public bool Exist(char[][] board, string word) 
    {
        int rows = board.Length;
        int cols = board[0].Length;
        bool[,] visited = new bool[rows, cols];

        for (int row = 0; row < rows; row++) 
        {
            for (int col = 0; col < cols; col++) 
            {
                if (DFS(board, row, col, rows, cols, 0, word, visited)) 
                {
                    return true;
                }
            }
        }

        return false;
    }

    private bool DFS(
        char[][] board, 
        int row, 
        int col, 
        int rows, 
        int cols, 
        int index, 
        string word, 
        bool[,] visited) 
    {
        if (row < 0 || row >= rows || col < 0 || col >= cols) 
        {
            return false;
        }

        if (visited[row, col] || board[row][col] != word[index]) 
        {
            return false;
        }

        if (index == word.Length - 1) 
        {
            return true;
        }

        visited[row, col] = true;

        bool found = DFS(board, row - 1, col, rows, cols, index + 1, word, visited) ||
                     DFS(board, row + 1, col, rows, cols, index + 1, word, visited) ||
                     DFS(board, row, col - 1, rows, cols, index + 1, word, visited) ||
                     DFS(board, row, col + 1, rows, cols, index + 1, word, visited);

        visited[row, col] = false;

        return found;
    }
}
```

**Time Complexity**

- **$O(M \cdot N \cdot 3^L)$** where $M$ is the number of rows, $N$ is the number of columns, and $L$ is the length of `word`.
    
    - The nested loops visit each of the $M \cdot N$ cells as a potential starting point.
        
    - From the starting cell, the search branches in up to 4 directions initially, and then at most 3 directions for subsequent steps (since backtracking prevents immediately revisiting the cell you came from). The max recursion depth is bounded by $L$.
        

**Space Complexity**

- **$O(M \cdot N + L)$**
    
    - **$O(M \cdot N)$** auxiliary space allocated for the 2D `visited` array.
        
    - **$O(L)$** call stack space for the recursive DFS, which grows up to the length of the string `word`.
