Question: https://leetcode.com/problems/wildcard-matching/

```cs
public class Solution 
{
    public bool IsMatch(string s, string p) 
    {
        int rows = s.Length;
        int cols = p.Length;

        bool[,] dp = new bool[rows + 1, cols + 1];

        // Base case: empty pattern matches empty string
        dp[rows, cols] = true;

        // Base case: empty pattern matching non-empty string is false
        for (int row = 0; row < rows; row++)
        {
            dp[row, cols] = false;
        }

        // Base case: empty string matching trailing '*' in pattern
        for (int col = cols - 1; col >= 0; col--)
        {
            if (p[col] == '*')
            {
                // The empty string can match the remaining pattern if and only if the rest of that pattern can match the empty string.
                dp[rows, col] = dp[rows, col + 1];
            }
            else
            {
                dp[rows, col] = false;
            }
        }

        // Bottom-up Dynamic Programming
        for (int row = rows - 1; row >= 0; row--)
        {
            for (int col = cols - 1; col >= 0; col--)
            {
                if (p[col] == '*')
                {
                    // '*' can match 0 characters (dp[row, col+1]) or 1+ characters (dp[row+1, col])
                    // dp[row, col + 1] represents 0 characters
                    // dp[row+1, col] respresents 1+ characters
                    dp[row, col] = dp[row, col + 1] || dp[row + 1, col];
                }
                else if (p[col] == '?' || s[row] == p[col])
                {
                    dp[row, col] = dp[row + 1, col + 1];
                }
                else
                {
                    dp[row, col] = false;
                }
            }
        }

        return dp[0, 0];
    }
}
```
