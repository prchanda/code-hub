### Recursive version

```cs
public class Solution 
{
    public bool WordBreak(string s, IList<string> wordDict) 
    {
        HashSet<string> lookup = new HashSet<string>(wordDict);        
        return Solve(s, lookup, 0);
    }

    private bool Solve(string s, HashSet<string> lookup, int index)
    {
        for (int length = 1; length <= s.Length - index; length++)
        {
            if (lookup.Contains(s.Substring(index, length)))
            {
                if (index + length == s.Length || Solve(s, lookup, index + length))
                {
                    return true;
                }
            }
        }

        return false;
    }
}
```


### Top-Down approach

```cs
public class Solution 
{
    public bool WordBreak(string s, IList<string> wordDict) 
    {
        HashSet<string> lookup = new HashSet<string>(wordDict);
        bool?[] cache = new bool?[s.Length + 1];        
        cache[s.Length] = true;
        return Solve(s, lookup, 0, cache);
    }

    private bool Solve(string s, HashSet<string> lookup, int index, bool?[] cache)
    {
        if (cache[index].HasValue)
        {
            return cache[index].Value;
        }
        
        for (int length = 1; length <= s.Length - index; length++)
        {
            if (lookup.Contains(s.Substring(index, length)))
            {
                if (Solve(s, lookup, index + length, cache))
                {                    
                    cache[index] = true;
                    return cache[index].Value;
                }
            }
        }

        cache[index] = false;
        return cache[index].Value;
    }
}
```

### Bottom-up approach

```cs
public class Solution 
{
    public bool WordBreak(string s, IList<string> wordDict) 
    {
        bool[] dp = new bool[s.Length + 1];
        dp[s.Length] = true;

        for (int index = s.Length - 1; index >= 0; index--)
        {
            foreach (var word in wordDict)
            {
                // Ensure sufficient characters remain in 's' and compare substring with 'word'
                if (index + word.Length<=s.Length && s.Substring(index, word.Length) == word)
                {
                    dp[index] = dp[index + word.Length];
                }

                // Early exit if a valid segmentation is found for the current index
                if (dp[index])
                {
                    break;
                }
            }
        }

        return dp[0];
    }
}
```

## ### Complexity Analysis

Let $N$ be the length of the string `s`, $M$ be the number of words in `wordDict`, and $K$ be the maximum length of a word in `wordDict`.

- **Time Complexity:** $O(N \cdot M \cdot K)$
    
    - The outer loop runs $N$ times (from $N-1$ down to $0$).
        
    - The inner loop iterates over all $M$ words in `wordDict`.
        
    - For each word, `s.Substring(index, word.Length)` extracts a string of length at most $K$ and performs a string equality comparison, taking $O(K)$ time.
        
    - Overall Time Complexity: $O(N \cdot M \cdot K)$.
        
- **Space Complexity:** $O(N)$
    
    - **DP Array:** The `dp` boolean array requires $O(N)$ auxiliary space.
        
    - **Substring Allocation:** Each call to `s.Substring(...)` temporarily allocates $O(K)$ space per check, which is reclaimed during garbage collection.

### HashSet Optimization — $O(N \cdot K^2)$

Instead of iterating over every word in `wordDict` ($M$), iterate only up to the maximum word length ($K$) from the current index. Lookups are done using a `HashSet`.

```cs
using System;
using System.Collections.Generic;

public class Solution 
{
    public bool WordBreak(string s, IList<string> wordDict) 
    {
        HashSet<string> wordLookup = new HashSet<string>(wordDict);
        bool[] dp = new bool[s.Length + 1];
        dp[s.Length] = true;

        // Find max length to limit sub-problems
        int maxLen = 0;
        foreach (var word in wordDict)
        {
            maxLen = Math.Max(maxLen, word.Length);
        }

        for (int i = s.Length - 1; i >= 0; i--)
        {
            for (int len = 1; len <= maxLen && i + len <= s.Length; len++)
            {
	            // `dp[i + len]` tells us. Can the rest of the string after that word be broken successfully?
                if (dp[i + len] && wordLookup.Contains(s.Substring(i, len)))
                {
                    dp[i] = true;
                    break;
                }
            }
        }

        return dp[0];
    }
}
```


- **Time Complexity:** $O(N \cdot K^2)$ — Outer loop runs $N$ times, inner loop runs up to $K$ times, and `s.Substring(i, len)` takes $O(K)$ time to slice and hash.
    
- **Space Complexity:** $O(N + W)$ — $O(N)$ for the `dp` array plus $O(W)$ space to store dictionary words in the `HashSet`.
