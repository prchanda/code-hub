Question: https://leetcode.com/problems/edit-distance/

`dp[i][j]` = minimum operations required to convert the first `i` characters of word1 into the first `j` characters of word2.

Explanation: https://neetcode.io/solutions/edit-distance

### Recursive version

```cs
public class Solution 
{
    public int MinDistance(string word1, string word2) 
    {
        return Solve(word1, word2, 0, 0);
    }

    private int Solve(string word1, string word2, int i, int j)
    {
        if (i == word1.Length)
        {
            return word2.Length - j;
        }
        
        if (j == word2.Length)
        {
            return word1.Length - i;
        }
        
        if (word1[i] == word2[j])
        {
            return Solve(word1, word2, i + 1, j + 1);
        }

        int delete = 1 + Solve(word1, word2, i + 1, j);
        int insert = 1 + Solve(word1, word2, i, j + 1);
        int replace = 1 + Solve(word1, word2, i + 1, j + 1);

        return Math.Min(delete, Math.Min(insert, replace));
    }
}
```


### Bottom-up approach

```cs
public class Solution 
{
    public int MinDistance(string word1, string word2) 
    {
        int rows = word1.Length;
        int cols = word2.Length;

        int[,] minDistance = new int[rows + 1, cols + 1];

        // Base cases: filling the last column and last row
        for (int row = 0; row <= rows; row++) 
        {
            minDistance[row, cols] = rows - row;
        }

        for (int col = 0; col <= cols; col++) 
        {
            minDistance[rows, col] = cols - col;
        }

        // Bottom-up Dynamic Programming
        for (int row = rows - 1; row >= 0; row--) 
        {
            for (int col = cols - 1; col >= 0; col--) 
            {
                if (word1[row] == word2[col]) 
                {
                    minDistance[row, col] = minDistance[row + 1, col + 1];
                } 
                else 
                {
                    int deleteCost = minDistance[row + 1, col];
                    int insertCost = minDistance[row, col + 1];
                    int replaceCost = minDistance[row + 1, col + 1];

                    minDistance[row, col] = 1 + Math.Min(deleteCost, Math.Min(insertCost, replaceCost));
                }
            }
        }

        return minDistance[0, 0];
    }
}
```

### Complexity Analysis

Let $m$ be the length of `word1` (`rows`) and $n$ be the length of `word2` (`cols`).

- **Time Complexity:** $\mathcal{O}(m \times n)$
    
    - Filling the base cases takes $\mathcal{O}(m + n)$ operations.
        
    - The nested loops iterate $m \times n$ times, performing constant-time $\mathcal{O}(1)$ operations inside each iteration.
        
- **Space Complexity:** $\mathcal{O}(m \times n)$
    
    - The 2D array `minDistance` requires storage of size $(m + 1) \times (n + 1)$.
        

> **Tip:** You can optimize the space complexity down to $\mathcal{O}(\min(m, n))$ since computing the current row only requires values from the previous/next row in the table.

```cs
public class Solution
{
    public int MinDistance(string word1, string word2)
    {
        int rows = word1.Length;
        int cols = word2.Length;

        // dp[col] represents the current row's value.
        int[] dp = new int[cols + 1];

        // Base case: word1 is exhausted.
        // Distance = remaining characters in word2.
        for (int col = 0; col <= cols; col++)
        {
            dp[col] = cols - col;
        }

        for (int row = rows - 1; row >= 0; row--)
        {
            // dp[cols] for the current row corresponds to (rows - row)
            int prevDiagonal = dp[cols]; // Stores minDistance[row + 1, col + 1]

            // dp[cols] = rows - row
            dp[cols] = rows - row;

            for (int col = cols - 1; col >= 0; col--)
            {
                // Save the old dp[col].
                // It represents dp[row + 1][col].
                int below = dp[col]; // Save minDistance[row + 1, col] before overwriting

                if (word1[row] == word2[col])
                {
                    dp[col] = prevDiagonal;
                }
                else
                {
                    int deleteCost = below;       // minDistance[row + 1, col]
                    int insertCost = dp[col + 1]; // minDistance[row, col + 1]
                    int replaceCost = prevDiagonal;   // minDistance[row + 1, col + 1]

                    dp[col] = 1 + Math.Min(
                        deleteCost,
                        Math.Min(insertCost, replaceCost)
                    );
                }

                // Old dp[row + 1][col] becomes
                // diagonal for the next column.
                prevDiagonal = below;
            }
        }

        return dp[0];
    }
}
```
