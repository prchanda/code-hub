Question: https://leetcode.com/problems/maximal-square/description/

Explanation : https://neetcode.io/solutions/maximal-square

```cs
using System;

public class Solution
{
    private int maxLength;

    public int MaximalSquare(char[][] matrix)
    {
        if (matrix == null || matrix.Length == 0 || matrix[0].Length == 0) 
            return 0;

        int rows = matrix.Length;
        int cols = matrix[0].Length;

        // Reset state for new calls
        maxLength = 0;

        int[,] cache = new int[rows, cols];

        // -1 indicates "not calculated yet"
        for (int row = 0; row < rows; row++)
        {
            for (int col = 0; col < cols; col++)
            {
                cache[row, col] = -1;
            }
        }

        FillCache(matrix, 0, 0, rows, cols, cache);

        return maxLength * maxLength;
    }

    private int FillCache(
        char[][] matrix, 
        int row, 
        int col, 
        int rows, 
        int cols, 
        int[,] cache)
    {
        // Base case: Outside the matrix
        if (row >= rows || col >= cols)
            return 0;

        // Base case: Already calculated
        if (cache[row, col] != -1)
            return cache[row, col];

        int right = FillCache(matrix, row, col + 1, rows, cols, cache);
        int down = FillCache(matrix, row + 1, col, rows, cols, cache);
        int diag = FillCache(matrix, row + 1, col + 1, rows, cols, cache);

        if (matrix[row][col] == '1')
        {
            cache[row, col] = 1 + Math.Min(Math.Min(right, down), diag);
            maxLength = Math.Max(maxLength, cache[row, col]);
        }
        else
        {
            cache[row, col] = 0;
        }

        return cache[row, col];
    }
}
```

### Complexity Analysis

Let $m$ be the number of rows (`rows`) and $n$ be the number of columns (`cols`).

- **Time Complexity:** $\mathcal{O}(m \times n)$
    
    - Initializing the `cache` array with `-1` takes $\mathcal{O}(m \times n)$ operations.
        
    - Thanks to memoization, each state `(row, col)` is evaluated at most once.
        
    - Inside each cell evaluation, computing `right`, `down`, and `diag` transitions takes $\mathcal{O}(1)$ time.
        
- **Space Complexity:** $\mathcal{O}(m \times n)$
    
    - **Memoization Cache:** The 2D `cache` array requires $\mathcal{O}(m \times n)$ space.
        
    - **Recursion Stack:** The maximum depth of the call stack can reach $\mathcal{O}(m + n)$ along the diagonal traversal.
        

> **Tip:** This can also be solved using Bottom-Up Dynamic Programming to avoid call stack overhead, or further space-optimized to **$\mathcal{O}(n)$ space** using just a 1D array across row transitions.



**Bottom-Up approach**

```cs

using System;

public class Solution 
{
    public int MaximalSquare(char[][] matrix) 
    {
        if (matrix == null || matrix.Length == 0 || matrix[0].Length == 0)
            return 0;

        int rows = matrix.Length;
        int cols = matrix[0].Length;
        int maxSide = 0;

        // dp[r, c] represents the side length of the largest square 
        // whose TOP-LEFT corner is at matrix[r][c]
        int[,] dp = new int[rows + 1, cols + 1];

        // Iterating backwards from bottom-right to top-left
        for (int r = rows - 1; r >= 0; r--)
        {
            for (int c = cols - 1; c >= 0; c--)
            {
                if (matrix[r][c] == '1')
                {
                    dp[r, c] = 1 + Math.Min(dp[r + 1, c + 1], 
                                   Math.Min(dp[r + 1, c], dp[r, c + 1]));

                    maxSide = Math.Max(maxSide, dp[r, c]);
                }
            }
        }

        return maxSide * maxSide;
    }
}
```
