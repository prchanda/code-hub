Question: https://leetcode.com/problems/minimum-window-substring/description/

Explanation: https://www.youtube.com/watch?v=jSto0O4AJbM

```cs
public class Solution {
    public string MinWindow(string s, string t) {
        int left = 0, right = 0, have = 0, need = 0;
        Dictionary<char, int> tLookup = new Dictionary<char, int>();
        Dictionary<char, int> sLookup = new Dictionary<char, int>();        

        foreach(char c in t)
        {
            tLookup[c] = tLookup.GetValueOrDefault(c) + 1;
        }
        need = tLookup.Count;
        int maxLength = int.MaxValue;
        int[] result = new int[2]{-1, -1};
        
        while(right < s.Length)
        {
            char ch = s[right];
            if (tLookup.ContainsKey(ch))
            {
                sLookup[ch] = sLookup.GetValueOrDefault(ch) + 1;
                if (sLookup[ch] == tLookup[ch])
                    have++;
            }
            
            while(have == need)
            {
                if(right - left + 1 < maxLength)
                {
                    maxLength = right - left + 1;
                    (result[0], result[1]) = (left, right);
                }
                char leftChar = s[left];
                if (tLookup.ContainsKey(leftChar))
                {
                    sLookup[leftChar]--;
                    if (sLookup[leftChar] < tLookup[leftChar])
                        have--;
                }
                left++;      
            }
            right++;
        }
        return result[0] == -1 ? "" : s.Substring(result[0], result[1] - result[0] + 1);
    }
}
```

### Complexity Analysis

- **Time Complexity:** $\mathcal{O}(M + N)$, where $N$ is the length of string `s` and $M$ is the length of string `t`. Building the frequency map for `t` takes $\mathcal{O}(M)$ time, and the sliding window traverses `s` with `left` and `right` pointers moving at most $N$ steps each, making dictionary lookups and updates $\mathcal{O}(1)$ on average.
    
- **Space Complexity:** $\mathcal{O}(M + N)$ in the worst case (or $\mathcal{O}(\Sigma)$ where $\Sigma$ is the alphabet size), as the dictionaries store the unique characters of `t` and up to the length of the current window in `s`.
