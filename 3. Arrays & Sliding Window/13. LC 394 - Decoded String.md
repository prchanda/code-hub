Question: https://leetcode.com/problems/decode-string/description/

Explanation: https://www.youtube.com/watch?v=qB0zZpBJlh8


```cs
using System;
using System.Collections.Generic;
using System.Text;

public class Solution 
{
    public string DecodeString(string s) 
    {
        Stack<int> countStack = new Stack<int>();
        Stack<StringBuilder> stringStack = new Stack<StringBuilder>();

        StringBuilder current = new StringBuilder();
        int number = 0;

        foreach (char ch in s) 
        {
            if (char.IsDigit(ch)) 
            {
                // Handle multi-digit numbers
                number = number * 10 + (ch - '0');
            } 
            else if (ch == '[') 
            {
                // Save the current state
                countStack.Push(number);
                stringStack.Push(current);

                // Start a new nested string
                number = 0;
                current = new StringBuilder();
            } 
            else if (ch == ']') 
            {
                // Get the state of the previous level
                int count = countStack.Pop();
                StringBuilder previous = stringStack.Pop();

                // Append current string 'count' times
                for (int i = 0; i < count; i++) 
                {
                    previous.Append(current);
                }

                // Current becomes the decoded string including the previous level
                current = previous;
            } 
            else 
            {
                // Normal character
                current.Append(ch);
            }
        }

        return current.ToString();
    }
}
```

### Complexity

- **Time Complexity:** $\mathcal{O}(N)$, where $N$ is the length of the final decoded output string. Every character is processed and appended proportionally to its expansion size.
    
- **Space Complexity:** $\mathcal{O}(M)$, where $M$ is the maximum nesting depth of brackets and the size of intermediate strings stored across the stacks.
