Question: https://leetcode.com/problems/all-nodes-distance-k-in-binary-tree/description/

Explanation : https://www.youtube.com/watch?v=i9ORlEy6EsI

```cs
public class Solution 
{
    public IList<int> DistanceK(TreeNode root, TreeNode target, int k) 
    {
        Dictionary<TreeNode, TreeNode> parentLookup = new Dictionary<TreeNode, TreeNode>();
        parentLookup[root] = null;
        DFS(root, parentLookup);

        Queue<TreeNode> queue = new Queue<TreeNode>();
        HashSet<TreeNode> visited = new HashSet<TreeNode>();

        queue.Enqueue(target);
        visited.Add(target);

        // BFS outward from the target node
        while (queue.Count > 0 && k > 0)
        {
            int size = queue.Count;

            for (int index = 0; index < size; index++)
            {
                var currentNode = queue.Dequeue();
                var parentNode = parentLookup.GetValueOrDefault(currentNode, null);

                // Explore Parent
                if (parentNode != null && visited.Add(parentNode))
                {
                    queue.Enqueue(parentNode);
                }

                // Explore Left Child
                if (currentNode.left != null && visited.Add(currentNode.left))
                {
                    queue.Enqueue(currentNode.left);
                }

                // Explore Right Child
                if (currentNode.right != null && visited.Add(currentNode.right))
                {
                    queue.Enqueue(currentNode.right);
                }
            }

            k--;
        }

        // Collect remaining nodes at distance K
        IList<int> result = new List<int>();
        while (queue.Count > 0)
        {
            result.Add(queue.Dequeue().val);
        }

        return result; 
    }

    private void DFS(TreeNode root, Dictionary<TreeNode, TreeNode> parentLookup)
    {
        if (root == null)
            return;

        if (root.left != null)
            parentLookup[root.left] = root;

        if (root.right != null)
            parentLookup[root.right] = root;

        DFS(root.left, parentLookup);
        DFS(root.right, parentLookup);
    }
}
```

### Complexity Analysis

Let $N$ be the total number of nodes in the binary tree.

- **Time Complexity:** $\mathcal{O}(N)$
    
    - **DFS Phase:** Visits every node once to map parent pointers, taking $\mathcal{O}(N)$ time.
        
    - **BFS Phase:** Treats the tree as an undirected graph. Each node is enqueued and dequeued at most once, taking $\mathcal{O}(N)$ time.
        
    - **Total Time:** $\mathcal{O}(N)$.
        
- **Space Complexity:** $\mathcal{O}(N)$
    
    - **DFS Call Stack:** $\mathcal{O}(H)$ where $H$ is the tree height ($\mathcal{O}(\log N)$ for balanced trees, $\mathcal{O}(N)$ for skewed trees).
        
    - **`parentLookup` Hash Table:** Stores $N$ entries, taking $\mathcal{O}(N)$ space.
        
    - **`visited` Set & `queue` Queue:** Store up to $N$ nodes during the BFS traversal, taking $\mathcal{O}(N)$ space.
        
    - **Total Space:** $\mathcal{O}(N)$.
