Question: https://leetcode.com/problems/word-search-ii/description/

Explanation: https://www.youtube.com/watch?v=asbcE9mZz_U&t=995s

```cs
public class TrieNode
{
    public Dictionary<char, TrieNode> Children = new Dictionary<char, TrieNode>();
    public string Word { get; set; }
}

public class Solution 
{
    public List<string> FindWords(char[][] board, string[] words) 
    {
        TrieNode root = new TrieNode();

        // Build Trie
        foreach (string word in words)
        {
            TrieNode node = root;

            foreach (char ch in word)
            {
                if (!node.Children.TryGetValue(ch, out TrieNode next))
                {
                    next = new TrieNode();
                    node.Children[ch] = next;
                }

                node = next;
            }

            // Store the complete word at the terminal node.
            node.Word = word;
        }

        List<string> result = new List<string>();

        for (int row = 0; row < board.Length; row++)
        {
            for (int col = 0; col < board[0].Length; col++)
            {
                DFS(board, row, col, root, result);
            }
        }

        return result;
    }

    private void DFS(char[][] board, int r, int c, TrieNode node, List<string> result)
    {
        if (r < 0 || r >= board.Length ||
            c < 0 || c >= board[0].Length)
        {
            return;
        }

        char ch = board[r][c];

        if (ch == '#' || !node.Children.ContainsKey(ch))
        {
            return;
        }

        node = node.Children[ch];

        if (node.Word != null)
        {
            result.Add(node.Word);
            /* To prevent the same word from being added to the result multiple times.
            board =
            [
                ['o','a'],
                ['a','t']
            ]
            words = ["oat"]
            */
            node.Word = null;
        }
        
        board[r][c] = '#';

        DFS(board, r - 1, c, node, result);
        DFS(board, r + 1, c, node, result);
        DFS(board, r, c - 1, node, result);
        DFS(board, r, c + 1, node, result);

        // backtrack as we have find from other positions
        board[r][c] = ch;
    }
}
```

**Complexity Analysis**

Let $M \times N$ be the grid dimensions (`board.Length` $\times$ `board[0].Length`), $W$ be the total number of words in `words`, and $L$ be the maximum length of a word.

| **Metric**           | **Complexity**                       | **Explanation**                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                         |
| -------------------- | ------------------------------------ | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Time Complexity**  | $O(W \cdot L + M \cdot N \cdot 3^L)$ | $O(W \cdot L)$ time to insert all words into the Trie. DFS starts from $M \cdot N$ grid positions and explores up to depth $L$, with a branching factor of at most 3 per move (since we cannot backtrack to the immediate previous cell).<br><br>Why `3^L`? This is the interesting part. Imagine we're currently at:<br><br>        ↑<br>   ←  X  →<br>        ↓<br><br>There are initially **4 directions**. But after moving into one cell, we cannot immediately go back to the cell we just came from because that cell is marked `#`. So after the first move, there are at most: 3  choices.<br><br>        start<br>       /  \|  \<br>      3   3   3<br>     /\|\ /\|\ /\|\<br>    ...<br><br>	Therefore, roughly:<br><br>depth 1 → 3<br>depth 2 → 3²<br>depth 3 → 3³<br>...<br>depth L → 3^L<br><br>So one starting cell can explore up to: O(3^L) paths<br> |
| **Space Complexity** | $O(W \cdot L)$                       | The Trie structure stores up to $W \cdot L$ total nodes across all words. The maximum depth of the call stack during DFS is bounded by $O(L)$.                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                          |
