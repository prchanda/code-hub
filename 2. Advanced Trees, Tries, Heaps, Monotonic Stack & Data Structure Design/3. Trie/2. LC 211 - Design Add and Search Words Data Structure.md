```cs

public class TrieNode
{
    public Dictionary<char, TrieNode> Children { get; } = new Dictionary<char, TrieNode>();
    public bool IsWord { get; set; }
}

public class WordDictionary
{
    private readonly TrieNode root;

    public WordDictionary()
    {
        root = new TrieNode();
    }

    public void AddWord(string word)
    {
        TrieNode node = root;

        foreach (char ch in word)
        {
            if (!node.Children.ContainsKey(ch))
            {
                node.Children[ch] = new TrieNode();
            }

            node = node.Children[ch];
        }

        node.IsWord = true;
    }

    public bool Search(string word)
    {
        return DFS(0, word, root);
    }

    private bool DFS(int index, string word, TrieNode node)
    {
        for (int i = index; i < word.Length; i++)
        {
            char ch = word[i];

            if (ch == '.')
            {
                foreach (TrieNode child in node.Children.Values)
                {
                    if (DFS(i + 1, word, child))
                    {
                        return true;
                    }
                }

                return false;
            }

            if (!node.Children.ContainsKey(ch))
            {
                return false;
            }

            node = node.Children[ch];
        }

        return node.IsWord;
    }
}
```

### Complexity Analysis

Let $N$ be the length of the string `word`, $\Sigma$ be the alphabet size (e.g., $26$ for lowercase English letters), and $M$ be the total number of nodes across all stored words in the Trie.

| **Operation**                    | **Time Complexity**                | **Space Complexity**                          |
| -------------------------------- | ---------------------------------- | --------------------------------------------- |
| **`AddWord`**                    | $O(N)$                             | $O(N)$ worst-case (if all characters are new) |
| **`Search` (Without Wildcards)** | $O(N)$                             | $O(N)$ call stack depth                       |
| **`Search` (With Wildcards)**    | $O(M)$ or $O(\Sigma^N)$ worst-case | $O(N)$ call stack depth                       |

**Detailed Breakdown**

- **`AddWord` Time:** $O(N)$ — The algorithm iterates through each character of the word once. Dictionary insertion and lookup operation takes $O(1)$ average time.
    
- **`Search` Time:**
    
    - **Best/Average Case (Exact Match):** $O(N)$ — Navigates directly down the Trie path of length $N$.
        
    - **Worst Case (Wildcards):** $O(\Sigma^N)$ or bounded by $O(M)$ — When searching patterns like `....`, the recursion must branch to explore all children at each node, potentially visiting every node stored in the Trie.
        
- **Overall Memory (Data Structure):** $O(M \cdot \Sigma)$ worst-case space to hold $M$ total characters across all added words alongside dictionary overhead.
