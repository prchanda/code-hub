Question: https://leetcode.com/problems/distribute-coins-in-binary-tree/description/

Explanation: https://www.youtube.com/watch?v=YfdfIOeV_RU

```cs
public class Solution {
    public int DistributeCoins(TreeNode root) {
        int moves = 0;
        DFS(root, ref moves);
        return moves;
    }

    private int DFS(TreeNode root, ref int moves) {
        if (root == null)
            return 0;

        int left = DFS(root.left, ref moves);
        int right = DFS(root.right, ref moves);
        moves += Math.Abs(left) + Math.Abs(right);

        return left + right + root.val - 1;        
    }
}
```
