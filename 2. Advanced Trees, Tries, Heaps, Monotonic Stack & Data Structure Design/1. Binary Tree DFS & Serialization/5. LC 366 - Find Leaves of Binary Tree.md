Given the root of a binary tree, collect a tree's nodes as if you were doing this:

1. Collect all the leaf nodes.
    
2. Remove all the leaf nodes.
    
3. Repeat until the tree is empty.
    

Return a list of all the values corresponding to these leaf collections.

### Example

**Input:**

Plaintext

```
      1
     / \
    2   3
   / \   
  4   5  
```

**Output:**

Plaintext

```
[[4, 5, 3], [2], [1]]
```

**Explanation:**

- First, the leaves are `4`, `5`, and `3`. Removing them leaves `2` with its left and right children as null.
    
- Next, the leaf is `2`. Removing it leaves `1`.
    
- Finally, the leaf is `1`. The tree is now empty.
    
- The result groups the nodes removed in each step: `[[4, 5, 3], [2], [1]]`. (Note: Order within the inner arrays usually doesn't strictly matter as long as they represent the correct height/step, but left-to-right is standard).
    

### Constraints

- The number of nodes in the tree is in the range $[1, 100]$.
    
- $-100 \le \text{Node.val} \le 100$



```cs
public class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;
    TreeNode() {}
    TreeNode(int val) { this.val = val; }
    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}

class Solution
{
    public IList<IList<int>> FindLeaves(TreeNode root)
    {
        List<IList<int>> result = new List<IList<int>>();

        DFS(root, result);

        return result;
    }

    private int DFS(TreeNode node, List<IList<int>> result)
    {
        if (node == null)
            return -1;

        int leftHeight = DFS(node.left, result);
        int rightHeight = DFS(node.right, result);

        int height = Math.Max(leftHeight, rightHeight) + 1;

        if (result.Count == height)
            result.Add(new List<int>());

        result[height].Add(node.val);

        return height;
    }
}
```
