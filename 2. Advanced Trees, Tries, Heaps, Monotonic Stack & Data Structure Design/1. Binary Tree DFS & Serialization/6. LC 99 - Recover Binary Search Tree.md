Question : https://leetcode.com/problems/recover-binary-search-tree/description/

```cs
public class Solution
{
    public void RecoverTree(TreeNode root)
    {
        TreeNode first = null;
        TreeNode second = null;
        TreeNode prev = null;
        TreeNode current = root;

        while (current != null)
        {
            if (current.left == null)
            {
                // Visit current
                CheckViolation(prev, current, ref first, ref second);
                prev = current;
                current = current.right;
            }
            else
            {
                TreeNode predecessor = current.left;

                while (predecessor.right != null && predecessor.right != current)
                {
                    predecessor = predecessor.right;
                }

                if (predecessor.right == null)
                {
                    // Create Thread
                    predecessor.right = current;
                    current = current.left;
                }
                else
                {
                    // Visit current
                    CheckViolation(prev, current, ref first, ref second);
                    // Remove Thread
                    predecessor.right = null;                    
                    prev = current;
                    current = current.right;
                }
            }
        }

		// Swap the actual node values
        int temp = first.val;
        first.val = second.val;
        second.val = temp;
    }

    private void CheckViolation(TreeNode prev, TreeNode current, ref TreeNode first, ref TreeNode second)
    {
        if (prev != null && prev.val > current.val)
        {
            // first remembers the first node that is too large
            if (first == null)
            {
                first = prev;
            }
            // second remembers the latest node that is too small
            second = current;
        }
    }
}
```


### Complexity Analysis

- **Time Complexity:** $O(N)$, where $N$ is the number of nodes in the binary tree. Using Morris Traversal, each node's left subtree is threaded and visited, meaning every edge in the tree is traversed a constant number of times (at most twice: once to establish the thread and once to remove it).
    
- **Space Complexity:** $O(1)$ auxiliary space. The algorithm avoids using an explicit stack or recursion, modifying tree pointers temporarily to perform an in-place in-order traversal.


## Morris InOrder traversal

```cs
public IList<int> InorderTraversal(TreeNode root)
{
    var result = new List<int>();
    TreeNode current = root;

    while (current != null)
    {
        // Case 1: No left subtree
        if (current.left == null)
        {
            result.Add(current.val);
            current = current.right;
        }
        else
        {
            // Find inorder predecessor
            TreeNode predecessor = current.left;

            while (predecessor.right != null &&
                   predecessor.right != current)
            {
                predecessor = predecessor.right;
            }

            // Case 2: First time reaching current
            if (predecessor.right == null)
            {
                // Create temporary thread
                predecessor.right = current;

                // Go to left subtree
                current = current.left;
            }
            // Case 3: Returning through the temporary thread
            else
            {
                // Remove temporary thread
                predecessor.right = null;

                // Visit current
                result.Add(current.val);

                // Go to right subtree
                current = current.right;
            }
        }
    }

    return result;
}
```
