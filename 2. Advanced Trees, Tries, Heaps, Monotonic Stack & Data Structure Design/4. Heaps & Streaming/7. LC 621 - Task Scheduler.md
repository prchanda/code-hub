```cs
public class Solution
{
    public int LeastInterval(char[] tasks, int n)
    {
        // Count frequency of each task
        Dictionary<char, int> frequency = new Dictionary<char, int>();

        foreach (char task in tasks)
        {
            frequency[task] = frequency.GetValueOrDefault(task) + 1;
        }

        // Max-Heap: highest frequency comes first
        PriorityQueue<int, int> maxHeap =
            new PriorityQueue<int, int>(
                Comparer<int>.Create((a, b) => b.CompareTo(a))
            );

        foreach (var pair in frequency)
        {
            maxHeap.Enqueue(pair.Value, pair.Value);
        }

        // The queue helps in cooling down the task (remaining count, available time)
        Queue<(int count, int availableTime)> cooldown = new Queue<(int count, int availableTime)>();

        int time = 0;
        while (maxHeap.Count > 0 || cooldown.Count > 0)
        {
            time++;
            // Execute the task with the highest remaining frequency
            if (maxHeap.Count > 0)
            {
                int count = maxHeap.Dequeue();
                count--;
                if (count > 0)
                {
                    cooldown.Enqueue((count, time + n+1));
                }
            }

            // Move cooled-down task back into max heap
            if (cooldown.Count > 0 &&
                cooldown.Peek().availableTime <= time)
            {
                var task = cooldown.Dequeue();
                maxHeap.Enqueue(task.count, task.count);
            }
        }

        return time;
    }
}
```
